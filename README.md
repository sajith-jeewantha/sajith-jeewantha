<h2 align="center">💻 Code Showcase</h2>

**A small feed-forward neural network in PyTorch**

```python
import torch
import torch.nn as nn

class MLP(nn.Module):
    """A simple multi-layer perceptron for classification."""

    def __init__(self, in_features: int, hidden: int, num_classes: int):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(in_features, hidden),
            nn.ReLU(),
            nn.Dropout(0.2),              # light regularization
            nn.Linear(hidden, num_classes),
        )

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return self.net(x)                # raw logits; CrossEntropyLoss applies softmax

model = MLP(in_features=784, hidden=128, num_classes=10)
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
loss_fn = nn.CrossEntropyLoss()

x, y = torch.randn(32, 784), torch.randint(0, 10, (32,))  # one dummy batch
loss = loss_fn(model(x), y)
optimizer.zero_grad(); loss.backward(); optimizer.step()
print(f"loss: {loss.item():.4f}")
```

**A scikit-learn classification pipeline**

```python
from sklearn.datasets import load_iris
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import classification_report
from sklearn.model_selection import train_test_split
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler

# Load a small, well-known dataset
X, y = load_iris(return_X_y=True)
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.2, stratify=y, random_state=42
)

# Scaling + model in one pipeline keeps preprocessing consistent
clf = make_pipeline(
    StandardScaler(),
    RandomForestClassifier(n_estimators=100, random_state=42),
)
clf.fit(X_train, y_train)

# Evaluate on held-out data
print(classification_report(y_test, clf.predict(X_test)))
```
